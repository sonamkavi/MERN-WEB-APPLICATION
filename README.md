    <h3>Backend Setup</h3>
    <pre>
    cd mern-auth-backend
    npm install
    npm start
    </pre>
    
   <h3>Frontend Setup</h3>
    <pre>
    cd mern-auth-frontend
    npm install
    npm start
    </pre>
    
  <h2>API Endpoints</h2>
    <ul>
        <li><code>POST /register</code> - User Registration</li>
        <li><code>POST /login</code> - User Login</li>
        <li><code>GET /user</code> - Retrieve User Data</li>
